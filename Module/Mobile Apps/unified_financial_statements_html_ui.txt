// Make sure tab functions are available
if(typeof switchFinTab === 'undefined'){
  window.switchFinTab = function(tab, el){
    document.querySelectorAll('[id^="tab-"]').forEach(t => t.style.display = 'none');
    document.getElementById(`tab-${tab}`).style.display = 'block';
    document.querySelectorAll('.fin-tab').forEach(b => {
      b.style.color = 'var(--sub)';
      b.style.borderBottomColor = 'transparent';
    });
    el.style.color = 'var(--primary)';
    el.style.borderBottomColor = 'var(--primary)';
  };
}
