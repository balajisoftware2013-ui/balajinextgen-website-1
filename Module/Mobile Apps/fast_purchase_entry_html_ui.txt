<!-- SUPPORTING JAVASCRIPT -->
<script>
function updateSupplierDisplay(){
  const suppId = document.getElementById('supplierSelect').value;
  if(suppId){
    const supp = DB.suppliers.find(s => s.id === suppId);
    if(supp && supp.name) {
      // Supplier selected, ready to add items
      setTimeout(() => addLineItemFast(), 100);
    }
  }
}

// Auto-fill date
document.addEventListener('DOMContentLoaded', function(){
  const dateInput = document.getElementById('invDate');
  if(dateInput && !dateInput.value){
    dateInput.value = new Date().toISOString().slice(0,10);
  }
});
</script>
