/* ── HELPER: look up client DB IDs from DATABASE_REGISTRY ── */
function _getClientDbIds(clientId) {
  if (!clientId) return {};
  try {
    // DATABASE_REGISTRY sheet columns:
    // CLIENT_ID | CLIENT_NAME | MASTER_DB_ID | TXN_DB_ID | REPORT_DB_ID | FOLDER_ID | STATUS
    const masterSS  = SpreadsheetApp.openById('1FuNJ_XejE2ekYTnk71wXVZ79hRJgu7pmIA6fuE-Iu7I');
    const regSheet  = masterSS.getSheetByName('DATABASE_REGISTRY');
    if (!regSheet) return {};
    const rows    = regSheet.getDataRange().getValues();
    const headers = rows[0].map(h => h.toString().toUpperCase().trim());
    const cCol    = headers.indexOf('CLIENT_ID');
    const mCol    = headers.indexOf('MASTER_DB_ID');
    const tCol    = headers.indexOf('TRANSACTION_DB_ID');
    const rCol    = headers.indexOf('REPORT_DB_ID');
    const fCol    = headers.indexOf('FOLDER_ID');
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][cCol] && rows[i][cCol].toString().trim() === clientId) {
        return {
          master:  mCol >= 0 ? rows[i][mCol].toString().trim() : '',
          txn:     tCol >= 0 ? rows[i][tCol].toString().trim() : '',
          report:  rCol >= 0 ? rows[i][rCol].toString().trim() : '',
          folder:  fCol >= 0 ? rows[i][fCol].toString().trim() : '',
        };
      }
    }
  } catch(e) {
    Logger.log('[_getClientDbIds] Error: ' + e.message);
  }
  return {};
}
